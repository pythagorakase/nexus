"""Environment and transcript capture for the isolated 812 manual proof."""
from pathlib import Path
import json
import os
import subprocess
import time

ROOT = Path(__file__).resolve().parents[2]
SCRATCH = ROOT / 'scratchpad/812-S3'
EVIDENCE = ROOT / 'docs/qa/812-models-plan-fetch'
PY = '/Users/pythagor/nexus/.venv/bin/python'


def proof_env(*, online=False):
    env = dict(os.environ)
    for key in list(env):
        if key.startswith(('HF_', 'HUGGINGFACE_')) or key in {
            'TRANSFORMERS_CACHE', 'TRANSFORMERS_OFFLINE', 'NEXUS_HOME',
            'NEXUS_RUNTIME_CONFIG', 'NEXUS_TEST_RECEIPTS_DIR', 'NEXUS_GATEWAY_PORT',
            'NEXUS_API_URL', 'NEXUS_SLOT', 'NEXUS_RUN_LIVE_LLM',
            'NEXUS_RUN_SECRET_STORE', 'NEXUS_RUN_CORPUS', 'NEXUS_TEST_PROVIDER_ONLY',
            'NEXUS_TEST_PROVIDER_DATABASE', 'NEXUS_DEV_DASHBOARD',
        }:
            env.pop(key)
    env.update(PYTHONPATH=str(ROOT), NEXUS_KEYRING_DISABLE='1',
               NEXUS_TEST_RECEIPTS_DIR=str(SCRATCH/'manual-receipts'),
               HF_HOME=str(SCRATCH/'fetch/hf-home'),
               HF_HUB_DISABLE_IMPLICIT_TOKEN='1', HF_DEBUG='0')
    if not online:
        env['HF_HUB_OFFLINE']='1'
    return env


def run(label, args, *, env=None, timeout=570):
    if os.getloadavg()[0] >= 24:
        raise RuntimeError('Load cap reached before '+label)
    command=['nice','-n','15',PY,*args]
    start=time.monotonic()
    with (EVIDENCE/f'{label}.stdout.log').open('w') as out, (EVIDENCE/f'{label}.stderr.log').open('w') as err:
        result=subprocess.run(command,cwd=ROOT,env=env or proof_env(),stdout=out,stderr=err,timeout=timeout)
    record={'label':label,'command':command,'returncode':result.returncode,
            'elapsed_seconds':round(time.monotonic()-start,3)}
    with (EVIDENCE/'manual-results.jsonl').open('a') as log:
        log.write(json.dumps(record)+'\n')
    print(json.dumps(record),flush=True)
    return result
