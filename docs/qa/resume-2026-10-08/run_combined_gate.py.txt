"""Run the documented three-piece PostgreSQL gate serially at one clean head."""

import datetime
import json
import os
from pathlib import Path
import subprocess
import sys


ROOT = Path('/Users/pythagor/.codex/worktrees/resume-claude-integration/nexus')
OUT = Path('/Users/pythagor/nexus/temp/orders_2026_10_08_codex/coord/full-gate')
PYTHON = '/Users/pythagor/nexus/.venv/bin/python'
PIECES = [
    ('core', ['tests', '--ignore=tests/test_api', '--ignore=tests/test_orrery']),
    ('api', ['tests/test_api']),
    ('orrery', ['tests/test_orrery']),
]


def git(*args):
    return subprocess.check_output(['git', *args], cwd=ROOT, text=True).strip()


def main():
    if git('status', '--porcelain'):
        raise RuntimeError('Gate requires a clean integration checkout')
    head = git('rev-parse', 'HEAD')
    OUT.mkdir(parents=True, exist_ok=False)
    env = dict(os.environ)
    for name in ('NEXUS_GATEWAY_PORT', 'NEXUS_API_URL', 'NEXUS_SLOT',
                 'NEXUS_RUN_LIVE_LLM', 'NEXUS_RUN_CORPUS'):
        env.pop(name, None)
    env.update(PYTHONPATH=str(ROOT), NEXUS_RUN_POSTGRES='1')
    results = {'head': head, 'started_at': datetime.datetime.now(datetime.timezone.utc).isoformat(),
               'pieces': [], 'state': 'running'}
    for name, selectors in PIECES:
        if git('rev-parse', 'HEAD') != head or git('status', '--porcelain'):
            raise RuntimeError('Integration checkout changed during gate')
        if os.getloadavg()[0] > 24:
            raise RuntimeError('Host load above the owner limit; gate stopped before next piece')
        command = ['nice', '-n', '15', PYTHON, '-m', 'pytest', '-q', '-p',
                   'tests.dbname_audit', '-p', 'no:cacheprovider', *selectors]
        log = OUT / f'{name}.log'
        item = {'name': name, 'command': command, 'log': str(log), 'state': 'running'}
        results['pieces'].append(item)
        (OUT / 'results.json').write_text(json.dumps(results, indent=2) + '\n')
        print(f'Starting {name} at {head}', flush=True)
        with log.open('w') as stream:
            completed = subprocess.run(command, cwd=ROOT, env=env, stdout=stream,
                                       stderr=subprocess.STDOUT, check=False)
        text = log.read_text()
        item.update(exit_code=completed.returncode,
                    secret_guard='secret-store guard: active; nexus-api: denied' in text,
                    owner_audit='dbname audit: owner targets: none' in text,
                    receipt_isolation='receipt isolation: checkout and user receipts untouched' in text,
                    state='complete')
        okay = (completed.returncode == 0 and item['secret_guard']
                and item['owner_audit'] and item['receipt_isolation'])
        print(f'Completed {name}: exit={completed.returncode}, guards={okay}', flush=True)
        if not okay:
            results['state'] = 'failed'
            (OUT / 'results.json').write_text(json.dumps(results, indent=2) + '\n')
            return 1
    results['state'] = 'passed'
    results['completed_at'] = datetime.datetime.now(datetime.timezone.utc).isoformat()
    (OUT / 'results.json').write_text(json.dumps(results, indent=2) + '\n')
    return 0


if __name__ == '__main__':
    sys.exit(main())
